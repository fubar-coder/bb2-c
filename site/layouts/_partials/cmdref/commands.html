{{- /* The Markdown of the page of a command library (acidlibs, userlibs). Context: dict "lib" (the data of the library, tools/cmdref), "intro" (the hand-written introduction, Markdown, may be empty) and "descs" (the hand-written descriptions of its commands from data/cmddesc, keyed by command name, may be empty), "fixes" (the rows of data/fixoverview.yaml that concern the library, may be empty) and "fixpage" (the library's page of fixed bugs, as a row's "page" names it). A fix is shown with every command its row names, and those that name no command of the library in a section of their own. Returns a string. */ -}} {{- $lib := .lib -}} {{- $descs := .descs | default dict -}} {{- $fixes := .fixes | default slice -}} {{- $md := "" -}} {{- with .intro -}} {{- $md = printf "%s\n\n" . -}} {{- end -}} {{- $facts := printf "Library number %v" $lib.number -}} {{- with $lib.object -}} {{- $facts = printf "%s. It manages `%s` objects, by default at most %v (the compiler options set the maximum)" $facts .name .maximum -}} {{- end -}} {{- /* the fixes of a command: the rows whose "command" names it */ -}} {{- $bycmd := dict -}} {{- $matched := dict -}} {{- range $lib.entries -}} {{- $name := lower .name -}} {{- $rows := slice -}} {{- range $fixes -}} {{- $row := . -}} {{- range split .command "," -}} {{- if eq (lower (trim (replaceRE `\s*\(.*\)$` "" .) " ")) $name -}} {{- $rows = $rows | append $row -}} {{- $matched = merge $matched (dict (string $row.issue) true) -}} {{- end -}} {{- end -}} {{- end -}} {{- with $rows -}} {{- $bycmd = merge $bycmd (dict $name (uniq .)) -}} {{- end -}} {{- end -}} {{- $other := slice -}} {{- range $fixes -}} {{- if not (index $matched (string .issue)) -}} {{- $other = $other | append . -}} {{- end -}} {{- end -}} {{- $fixline := "" -}} {{- with $fixes -}} {{- $n := len (uniq .) -}} {{- $where := cond $bycmd "below with the commands they concern" "at the end of this page" -}} {{- if eq $n 1 -}} {{- $where = cond $bycmd "below with the command it concerns" "at the end of this page" -}} {{- end -}} {{- if where . "page" $.fixpage -}} {{- $where = printf "%s and on the page [%s]({{< relref \"/docs/compared-with-original/fixed-bugs/%s\" >}}) of the fixed original bugs" $where (path.Base $.fixpage) $.fixpage -}} {{- end -}} {{- $fixline = printf "The BB2 C port fixes %d %s of the original in this library, listed %s." $n (cond (eq $n 1) "bug" "bugs") $where -}} {{- end -}} {{- if $lib.entries -}} {{- $md = printf "%s%s.\n\n" $md $facts -}} {{- with $fixline -}} {{- $md = printf "%s%s\n\n" $md . -}} {{- end -}} {{- $md = printf "%s## Commands\n\n" $md -}} {{- range sort $lib.entries "name" -}} {{- $md = printf "%s%s" $md (partial "cmdref/entry.html" (dict "entry" . "desc" (index $descs .name) "fixes" (index $bycmd (lower .name)))) -}} {{- end -}} {{- else -}} {{- $md = printf "%s%s.\n\nThis library declares no commands. Its routines are used by the compiler, by the code it generates or by other libraries.\n" $md $facts -}} {{- with $fixline -}} {{- $md = printf "%s\n%s\n" $md . -}} {{- end -}} {{- end -}} {{- with $other -}} {{- $md = printf "%s\n## Fixed original bugs\n\n%s\n\n%s" $md (cond $bycmd "The fixes of the library that concern no single command above:" "The fixes of the library:") (partial "cmdref/fixes.html" .) -}} {{- end -}} {{- return $md -}}
