{{- /* The Markdown of one entry of a command library: a statement, function, command, object name or keyword. Context: dict "entry" (the entry of the data, tools/cmdref) and "desc" (its hand-written description from data/cmddesc: "text" and optional "notes"; nil if there is none). Returns a string. */ -}} {{- $e := .entry -}} {{- $desc := .desc | default dict -}} {{- $notation := dict "byte" ".b" "word" ".w" "long" ".l" "quick" ".q" "float" ".f" "string" "$" "any" "any" -}} {{- $kinds := dict "statement" "Statement" "function" "Function" "command" "Statement or function" "object" "Object name" "keyword" "Keyword" -}} {{- $facts := slice (index $kinds $e.kind | default $e.kind) -}} {{- if $e.typed -}} {{- $facts = $facts | append "takes a type suffix such as `.w`" -}} {{- end -}} {{- with $e.result -}} {{- if eq . "suffix" -}} {{- $facts = $facts | append "result of the type of the suffix" -}} {{- else -}} {{- $facts = $facts | append (printf "result `%s`" (index $notation . | default .)) -}} {{- end -}} {{- end -}} {{- $md := printf "### %s\n\n" (partial "cmdref/escape.html" $e.name) -}} {{- with $desc.text -}} {{- $md = printf "%s%s\n\n" $md (strings.TrimSpace .) -}} {{- end -}} {{- $md = printf "%s%s." $md (delimit $facts ", ") -}} {{- if $e.help -}} {{- $md = printf "%s Help text: `%s`" $md $e.help -}} {{- else -}} {{- $md = printf "%s No help text." $md -}} {{- end -}} {{- with $e.forms -}} {{- if and (eq (len .) 1) (not (index . 0).args) -}} {{- $md = printf "%s\\\nNo arguments.\n\n" $md -}} {{- else if eq (len .) 1 -}} {{- $md = printf "%s\\\nArguments: %s\n\n" $md (partial "cmdref/form.html" (index . 0)) -}} {{- else -}} {{- $md = printf "%s\\\n%d forms:\n\n" $md (len .) -}} {{- range . -}} {{- $md = printf "%s- %s\n" $md (partial "cmdref/form.html" .) -}} {{- end -}} {{- $md = printf "%s\n" $md -}} {{- end -}} {{- else -}} {{- $md = printf "%s\n\n" $md -}} {{- end -}} {{- with $desc.notes -}} {{- $md = printf "%s%s\n\n" $md (strings.TrimSpace .) -}} {{- end -}} {{- return $md -}}
