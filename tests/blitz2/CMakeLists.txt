# Tests of blitz2's own behaviour where it is meant to differ from the
# original (an extension, named in its commit), so there is no reference
# to compare with: blitz2 compiles and runs ID.bas in Amiberry, and its
# result must equal ID.expect (format of tests/corpus/runner/bugorig.h).
# One corpusrun pass per test with expect= and regression=on (an expect
# test, the same as the port regression tests of tests/bug-orig/regression):
# no reference and nothing of the CD (no blitz2:defaultdbug), one
# configuration (clib: blitz2 with the C libraries) judged against the
# expectation, FIXED or REGRESSED.
#   ID.bas      the program (text, tokenised with TED's token list); its
#               header line "; @runerrs on|off" sets runtime error
#               checking (default off)
#   ID.expect   the expected result
#   ID.inc      (optional) an include as text, tokenised the same way and
#               staged as CORPUS:src/ID.inc
# ctest blitz2_ID (- as _), label emulator, skipped (77) without Amiberry.
# With a prev build (tests/README.md "The prev build") the source is also compiled by the
# previous commit's compiler, its executable must be the same (compiler_prev, or
# blitz2_ID_prev for a test with an include or runtime error checking).
# The intrinsics (intrinsic-*.bas) check the compiled code themselves: the
# program counts the JSR instructions (opcode $4EB9, every library call)
# between two labels around the statement and prints the count, so the
# expectation names the cases that call the library and those that do not.
# The self test selftest/ID.bas (an expectation the port does not meet on
# purpose, ctest blitz2_selftest_ID) must come out REGRESSED, with the
# mismatching case in the report.
set(BLITZ2_TEST_TIMEOUT 90 CACHE STRING "tests/blitz2: time limit of one emulator session (seconds)")
set(ted_tokens ${PROJECT_SOURCE_DIR}/tests/data/ted-tokens.txt)
set(blitz2_test_sources)

# one expect test: ID.bas and ID.expect in dir; name the ctest name;
# selftest: the expectation is wrong on purpose (exit status 1 wanted)
function(blitz2_test bas dir name selftest)
	get_filename_component(id ${bas} NAME_WE)
	set(expect ${dir}/${id}.expect)
	if(NOT EXISTS ${expect})
		message(FATAL_ERROR "tests/blitz2: ${bas} has no ${id}.expect")
	endif()
	set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${bas})
	# (the ";" of the header line is CMake's list separator: matched behind it)
	file(STRINGS ${bas} runerrs REGEX "^; @runerrs ")
	set(runtime_errors off)
	if(runerrs MATCHES "@runerrs (on|off)$")
		set(runtime_errors ${CMAKE_MATCH_1})
	elseif(runerrs)
		message(FATAL_ERROR "tests/blitz2: ${bas}: bad header line: ${runerrs}")
	endif()

	set(src ${CMAKE_CURRENT_BINARY_DIR}/src/${id}.bb2)
	add_custom_command(OUTPUT ${src}
		COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/src
		COMMAND ${TEDSRC} tok ${ted_tokens} ${bas} ${src}
		DEPENDS ${bas} ${ted_tokens} ${TEDSRC} host_tools
		COMMENT "Tokenising tests/blitz2/${id}.bas"
		VERBATIM)
	set(srcs ${src})
	# ID.inc (text, optional): an include, tokenised like ID.bas, which
	# the program reads as CORPUS:src/ID.inc
	set(incfile)
	if(EXISTS ${dir}/${id}.inc)
		set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${dir}/${id}.inc)
		set(inc ${CMAKE_CURRENT_BINARY_DIR}/src/${id}.inc)
		add_custom_command(OUTPUT ${inc}
			COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/src
			COMMAND ${TEDSRC} tok ${ted_tokens} ${dir}/${id}.inc ${inc}
			DEPENDS ${dir}/${id}.inc ${ted_tokens} ${TEDSRC} host_tools
			COMMENT "Tokenising tests/blitz2/${id}.inc"
			VERBATIM)
		list(APPEND srcs ${inc})
		set(incfile "file=src/${id}.inc|${inc}\n")
	endif()

	set(work ${CMAKE_CURRENT_BINARY_DIR}/${id})
	file(GENERATE OUTPUT ${work}.pass CONTENT
"# blitz2_${id}: generated by tests/blitz2/CMakeLists.txt from ${bas}
work=${work}
report=${work}.txt
runtime_errors=${runtime_errors}
timeout=${BLITZ2_TEST_TIMEOUT}
parallel=1
ted=$<TARGET_FILE:tedstub>
launcher=$<TARGET_FILE:reqtrap>
config=clib|output|$<TARGET_FILE:blitz2>|${BLITZ2_DIR}/deflibs|${BLITZ2_DIR}/acidlibs
source=${src}
expect=${expect}
regression=on
${incfile}")
	if(selftest)
		add_test(NAME ${name} COMMAND ${CMAKE_COMMAND} -DRUNNER=${CORPUSRUN} -DSUB=pass
			-DPASS=${work}.pass -DWANT=1 -P ${PROJECT_SOURCE_DIR}/tests/bug-orig/want.cmake)
	else()
		add_test(NAME ${name} COMMAND ${CORPUSRUN} pass ${work}.pass)
	endif()
	# one session, maybe waiting for a free emulator slot
	math(EXPR limit "${BLITZ2_TEST_TIMEOUT} + 60 + ${EMULATION_QUEUE_WAIT}")
	set_tests_properties(${name} PROPERTIES LABELS emulator SKIP_RETURN_CODE 77
		TIMEOUT ${limit})
	# with a prev build: the executable of the previous commit's compiler must be the same
	# (cmake/prev.cmake prev_compile_test; not for the self tests)
	if(NOT selftest)
		prev_compile_test(${name} ${src} RUNTIME_ERRORS ${runtime_errors} EXTRA_VAR incfile)
	endif()
	set(blitz2_test_sources ${blitz2_test_sources} ${srcs} PARENT_SCOPE)
endfunction()

file(GLOB tests CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.bas)
# the tests of the BASIC test model (ID.test.bas) are registered by their own step (below)
list(FILTER tests EXCLUDE REGEX "\\.test\\.bas$")
foreach(bas IN LISTS tests)
	get_filename_component(id ${bas} NAME_WE)
	string(REPLACE "-" "_" name "blitz2_${id}")
	blitz2_test(${bas} ${CMAKE_CURRENT_SOURCE_DIR} ${name} OFF)
endforeach()
file(GLOB selftests CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/selftest/*.bas)
list(FILTER selftests EXCLUDE REGEX "\\.test\\.bas$")
foreach(bas IN LISTS selftests)
	get_filename_component(id ${bas} NAME_WE)
	string(REPLACE "-" "_" name "blitz2_selftest_${id}")
	blitz2_test(${bas} ${CMAKE_CURRENT_SOURCE_DIR}/selftest ${name} ON)
	# the mismatching case is named in the report: the line the program
	# printed and the line the expectation wanted
	file(STRINGS ${bas} want REGEX "^; @want ")
	if(want MATCHES "@want (.+)$")
		set_tests_properties(${name} PROPERTIES PASS_REGULAR_EXPRESSION "${CMAKE_MATCH_1}")
	else()
		message(FATAL_ERROR "tests/blitz2: ${bas} has no line \"; @want REGEX\"")
	endif()
endforeach()

# The comparison of a test of the model with the prev build (prev_compile_model_test, corpusrun
# case --prev) with broken input, in a build without a prev build: the driver of
# selftest/model-prev.test.bas (one compile, model-prev, without runtime error checking) compiled
# by this build's compiler as "prev", or by hangstub (tests/corpus/amiga/hangstub.c, never ends: a
# prev compiler whose result differs; the session's time limit is the test's, 30 s):
#   blitz2_selftest_model_same_prev      the same compiler: passes
#   blitz2_selftest_model_differs_prev   hangstub: fails, naming the compile
#   blitz2_selftest_model_intended_prev  hangstub, the compile named: passes, "(intended)"
#   blitz2_selftest_model_stale_prev     the same compiler, the compile named: fails, STALE
#   blitz2_selftest_model_unknown_prev   the same compiler, model-prev-runerrs named (the test
#                                        has no cell with runtime error checking): fails, STALE
set(model_prev_src ${CMAKE_CURRENT_SOURCE_DIR}/selftest/model-prev.test.bas)
set(model_prev_drv ${CMAKE_CURRENT_BINARY_DIR}/model-drivers/model-prev.bb2)
basic_test_driver(${model_prev_src} ${model_prev_drv} ${CMAKE_CURRENT_SOURCE_DIR}/selftest/model-prev.test.toml)
add_custom_target(blitz2_selftest_model_prev_driver ALL DEPENDS ${model_prev_drv})
function(model_prev_selftest kind compiler intended regex)
	set(name blitz2_selftest_model_${kind}_prev)
	set(cmd ${CORPUSRUN} case --env ${BASIC_TEST_ENV} --source ${model_prev_src} --prev
		--driver ${model_prev_drv} --test ${name} --prev-blitz2 ${compiler} --prev-libs ${BLITZ2_DIR})
	if(intended)
		list(APPEND cmd --intended ${name}:${intended})
	endif()
	add_test(NAME ${name} COMMAND ${cmd})
	math(EXPR limit "30 * 2 + 120 + 2 * ${EMULATION_QUEUE_WAIT}")
	set_tests_properties(${name} PROPERTIES LABELS emulator SKIP_RETURN_CODE 77 TIMEOUT ${limit}
		PASS_REGULAR_EXPRESSION "${regex}")
endfunction()
model_prev_selftest(same $<TARGET_FILE:blitz2> ""
	"model-prev[^\n]*prev: result same, exe identical\n.*model-prev: the comparison with the prev build: the same\n")
model_prev_selftest(differs $<TARGET_FILE:hangstub> ""
	"model-prev[^\n]*no RESULT[^\n]*\nprev differs in: PREV_INTENDED=\"blitz2_selftest_model_differs_prev:model-prev\"\n.*model-prev: the comparison with the prev build: DIFFERS\n")
model_prev_selftest(intended $<TARGET_FILE:hangstub> model-prev
	"model-prev[^\n]*no RESULT[^\n]*\\(intended\\)\n.*model-prev: the comparison with the prev build: the same\n")
model_prev_selftest(stale $<TARGET_FILE:blitz2> model-prev
	"STALE blitz2_selftest_model_stale_prev:model-prev: intended, but prev does not differ.*model-prev: the comparison with the prev build: DIFFERS\n")
model_prev_selftest(unknown $<TARGET_FILE:blitz2> model-prev-runerrs
	"STALE blitz2_selftest_model_unknown_prev:model-prev-runerrs: intended, but the test has no such compile \\(its compiles: model-prev\\)\nmodel-prev: the comparison with the prev build: DIFFERS\n")
