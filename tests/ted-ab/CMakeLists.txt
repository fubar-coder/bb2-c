# TED A/B: the original TED against TED-C in Amiberry, driven by scripts
# (TODO.md item 5).
#   amiga/    the m68k tools: teddrive (plays a script against TED),
#             teddrivetest (its self-test, gate D), tedprobe (gate 0)
#   scripts/  the tests (NAME.drv, directives: runner/tedab.h) and the
#             gate D scripts (gate-*.drv); NAME.expect: text of @expect tok,
#             tokenised here with tedtok
#   src/      sources of @source E / S
#   lint/     scripts only for the lint of a directive (fixed: correct,
#             fixed-wrong: broken on purpose), never run
#   runner/   tedab (host side): runs the tests, see runner/main_tedab.c
#   host/     host tests of the tools (built by host/CMakeLists.txt)
add_subdirectory(amiga)

set(TED_TOKENS ${PROJECT_SOURCE_DIR}/tests/data/ted-tokens.txt)
set(TED_AB_EXPECT ${CMAKE_CURRENT_BINARY_DIR}/expect)
file(GLOB expects CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/scripts/*.expect)
set(ted_ab_tokenised)
foreach(e IN LISTS expects)
	get_filename_component(name ${e} NAME)
	set(out ${TED_AB_EXPECT}/${name}.bb2)
	add_custom_command(OUTPUT ${out}
		COMMAND ${CMAKE_COMMAND} -E make_directory ${TED_AB_EXPECT}
		COMMAND ${TEDTOK} ${TED_TOKENS} ${e} ${out}
		DEPENDS ${e} ${TED_TOKENS} ${TEDTOK} host_tools
		COMMENT "Tokenising scripts/${name}"
		VERBATIM)
	list(APPEND ted_ab_tokenised ${out})
endforeach()
add_custom_target(ted_ab_expect ALL DEPENDS ${ted_ab_tokenised})

# The pass: all combinations (R5 C5 Cc Rc, the reference first; P5 Pc with
# the prev build) unless
# TED_AB_COMBOS says otherwise, all tests unless TED_AB_FILTER (globs over
# the names, comma separated).  The T sources, the expectations "corpus"
# and the default .xtra come from the corpus (its tokenised tests and the
# ref run of ctest corpus: a fixture, ctest runs it first).
# TED_AB_BUDGET: estimated run time (@est) per session; boot and gates come
# on top, TED_AB_TIMEOUT is the session's limit.
set(TED_AB_COMBOS "R5,C5,Cc,Rc" CACHE STRING "ted_ab: the combinations, the reference first")
# with the prev build (BLITZ2_PREV, cmake/prev.cmake) P5 and Pc come on top:
# the TED of the previous commit with blitz2 2.10 and with the previous
# commit's compiler, compared with C5 and Cc
set(ted_ab_combos ${TED_AB_COMBOS})
set(ted_ab_prev)
if(PREV_REQUESTED)
	foreach(c P5 Pc)
		if(NOT ",${TED_AB_COMBOS}," MATCHES ",${c},")
			string(APPEND ted_ab_combos ",${c}")
		endif()
	endforeach()
	set(ted_ab_prev "ted_prev=${PREV_BLITZ2_DIR}/ted\nblitz2_prev=${PREV_BLITZ2_DIR}/blitz2\n")
endif()
set(TED_AB_FILTER "" CACHE STRING "ted_ab: only these tests (globs, comma separated)")
set(TED_AB_TIMEOUT 120 CACHE STRING "ted_ab: time limit of one emulator session (seconds)")
set(TED_AB_BUDGET 60 CACHE STRING "ted_ab: estimated run time per session (seconds)")
# the combinations of one test stay together in a session: with P5 and Pc it
# holds six runs instead of four, so budget and limit are one and a half times
set(ted_ab_timeout ${TED_AB_TIMEOUT})
set(ted_ab_budget ${TED_AB_BUDGET})
if(PREV_REQUESTED)
	math(EXPR ted_ab_timeout "${TED_AB_TIMEOUT} * 3 / 2")
	math(EXPR ted_ab_budget "${TED_AB_BUDGET} * 3 / 2")
endif()
# its sessions at the same time: more than the corpus passes (ted_ab is the
# longest test of the preset all; the emulator slots limit it anyway); 1
# where CORPUS_PARALLEL is 1 (no private control socket per session)
if(CORPUS_PARALLEL GREATER 1)
	set(ted_ab_parallel_default 5)
else()
	set(ted_ab_parallel_default 1)
endif()
set(TED_AB_PARALLEL ${ted_ab_parallel_default} CACHE STRING
	"ted_ab: emulator sessions at the same time")
set(work ${CMAKE_CURRENT_BINARY_DIR}/ted_ab)
set(text "# ted_ab: generated by tests/ted-ab/CMakeLists.txt\n")
string(APPEND text "work=${work}\nreport=${work}.txt\n"
	"scripts=${CMAKE_CURRENT_SOURCE_DIR}/scripts\nsrc=${CMAKE_CURRENT_SOURCE_DIR}/src\n"
	"combos=${ted_ab_combos}\n${ted_ab_prev}filter=${TED_AB_FILTER}\n"
	"corpus_src=${CMAKE_BINARY_DIR}/tests/corpus/src/tests\n"
	"corpus_out=${CMAKE_BINARY_DIR}/tests/corpus/corpus/ref/vol/out\n"
	"expect_dir=${TED_AB_EXPECT}\n"
	"cd=${PROJECT_SOURCE_DIR}/archive/cd/ultimate-blitz-basic-2.1\n"
	"oracle=${PROJECT_SOURCE_DIR}/tests/data/ted-oracle.sha256\n"
	"fonts=${PROJECT_SOURCE_DIR}/archive/sources/ted-1.1\n"
	"teddrive=$<TARGET_FILE:teddrive>\nteddrivetest=$<TARGET_FILE:teddrivetest>\n"
	"tedprobe=$<TARGET_FILE:tedprobe>\nreqtrap=$<TARGET_FILE:reqtrap>\n"
	"ted_ref=${CMAKE_BINARY_DIR}/programs/ted/ref/ted\nted_c=$<TARGET_FILE:ted>\n"
	"blitz2_ref=${PROJECT_SOURCE_DIR}/archive/cd/ultimate-blitz-basic-2.1/BLITZ2/BLITZ2\n"
	"blitz2_c=$<TARGET_FILE:blitz2>\n"
	"timeout=${ted_ab_timeout}\nbudget=${ted_ab_budget}\nparallel=${TED_AB_PARALLEL}\n")
file(GENERATE OUTPUT ${work}.pass CONTENT "${text}")

# ted_ab_lint: the scripts against teddrive's syntax, their directives and
# sources (no emulator); ted_ab: the runs
add_test(NAME ted_ab_lint COMMAND ${TEDAB} --lint ${work}.pass)
# the lint of the directive @fixed #NNN PAIR... (runner/tedab.h): lint/fixed
# a correct script (the sample of tests/README.md), lint/fixed-wrong scripts
# broken on purpose, each refused with its message
foreach(dir fixed fixed-wrong)
	file(GENERATE OUTPUT ${work}-lint-${dir}.pass
		CONTENT "${text}scripts=${CMAKE_CURRENT_SOURCE_DIR}/lint/${dir}\n")
endforeach()
add_test(NAME ted_ab_lint_fixed COMMAND ${TEDAB} --lint ${work}-lint-fixed.pass)
set_tests_properties(ted_ab_lint_fixed PROPERTIES
	PASS_REGULAR_EXPRESSION "^tedab: 1 tests, 0 problems\n$")
add_test(NAME ted_ab_lint_fixed_wrong COMMAND ${TEDAB} --lint ${work}-lint-fixed-wrong.pass)
set_tests_properties(ted_ab_lint_fixed_wrong PROPERTIES PASS_REGULAR_EXPRESSION
	"^a-no-pair line 4: @fixed #NNN PAIR[^\n]*\nb-bad-pair line 5: @fixed: C5-R5 is no compared pair[^\n]*\nc-not-run: @fixed #123 R5-C5: a combination not in @combos[^\n]*\nd-twice line 5: @fixed: pair R5-C5 named twice\ntedab: 0 tests, 4 problems\n$")
add_test(NAME ted_ab COMMAND ${TEDAB} ${work}.pass)
math(EXPR limit "3600 + ${EMULATION_QUEUE_WAIT}")
set_tests_properties(ted_ab PROPERTIES LABELS emulator SKIP_RETURN_CODE 77
	FIXTURES_REQUIRED corpus TIMEOUT ${limit})
