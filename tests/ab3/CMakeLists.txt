# The tests of blitz2-ab3 (docs/ab3/variant.md).
#
# Equivalence with blitz2 (BB2 mode): every source of the corpus
# (tests/corpus/tests*/) and of the bug-orig tests (tests/bug-orig/*/) is
# compiled through tedstub once by blitz2 (the reference) and once by
# blitz2-ab3, with the C libraries of this build and the bug-orig test
# library in blitz2:userlibs; the programs are not run (corpusrun
# run=off).  Every source must give the same compile result (tedstub's
# RESULT line: size, or error line and message) and the same executable -
# byte-identical without runtime error checking, with it the trap #1/#4
# parameter blocks zeroed (the address of the editor's line item depends
# on the memory layout).  A bug-orig source that needs more than that
# (RWE userlibs, its own jobs or residents) does not compile in either:
# it is compared by its error; two that cannot be compared are left out
# (below).  Label emulator, skipped (77) without
# Amiberry or the CD (blitz2:defaultdbug).
get_property(corpus_sources GLOBAL PROPERTY CORPUS_SOURCES)
get_property(bug_orig_sources GLOBAL PROPERTY BUG_ORIG_SOURCES)
get_property(bug_orig_userlib GLOBAL PROPERTY BUG_ORIG_USERLIB)
# not compared - NAME|reason:
#   compiler-incdir-overflow  the compile overwrites the data behind incdir
#                             (#16): the compiler hangs, the session ends
#   compiler-maxs-libjsr      the unresolved JSRs inside linked library code
#                             (#335) put addresses of the compiler's memory
#                             into the executable: they depend on its layout
foreach(n compiler-incdir-overflow compiler-maxs-libjsr)
	list(FILTER bug_orig_sources EXCLUDE REGEX "/${n}\\.bb2$")
endforeach()
set(CD_BLITZ2 ${PROJECT_SOURCE_DIR}/archive/cd/ultimate-blitz-basic-2.1/BLITZ2)
set(AB3_EQUIVALENCE_TIMEOUT 600 CACHE STRING
	"ab3 equivalence: time limit of one emulator session (seconds)")

# equivalence_pass(NAME on|off): pass file + ctest
function(equivalence_pass name runtime_errors)
	set(work ${CMAKE_CURRENT_BINARY_DIR}/${name})
	set(text "# ${name}: generated by tests/ab3/CMakeLists.txt\n")
	string(APPEND text "work=${work}\nreport=${work}.txt\nrun=off\n"
		"runtime_errors=${runtime_errors}\ntimeout=${AB3_EQUIVALENCE_TIMEOUT}\n"
		"parallel=${CORPUS_PARALLEL}\nted=$<TARGET_FILE:tedstub>\n"
		"defaultdbug=${CD_BLITZ2}/DEFAULTDBUG\n"
		"config=blitz2|ref|$<TARGET_FILE:blitz2>|${BLITZ2_DIR}/deflibs|${BLITZ2_DIR}/acidlibs\n"
		"config=ab3|exe|$<TARGET_FILE:blitz2-ab3>|${BLITZ2_DIR}/deflibs|${BLITZ2_DIR}/acidlibs\n"
		"file=blitz2/userlibs/bugorig/bugorig.obj|${bug_orig_userlib}\n")
	foreach(s IN LISTS corpus_sources bug_orig_sources)
		string(APPEND text "source=${s}\n")
	endforeach()
	file(GENERATE OUTPUT ${work}.pass CONTENT "${text}")
	add_test(NAME ${name} COMMAND ${CORPUSRUN} ${work}.pass)
	# two sessions, maybe one after the other, each maybe waiting for a slot
	math(EXPR limit "${AB3_EQUIVALENCE_TIMEOUT} * 2 + 120 + 2 * ${EMULATION_QUEUE_WAIT}")
	set_tests_properties(${name} PROPERTIES LABELS emulator SKIP_RETURN_CODE 77
		TIMEOUT ${limit})
endfunction()

equivalence_pass(ab3_bb2_equivalence off)
equivalence_pass(ab3_bb2_equivalence_runerr on)

# AB3 mode's token list (spec ab3-compat/libraries): blitz2-ab3's
# AmiBlitz3 list (the one ending with $80DB, programs/blitz2/src/toklist.h)
# followed by the tokens of AmiBlitz3 v3.14.0's BigLibs in its load order
# must equal the list of AmiBlitz3 v3.14.0's own compiler (the binary
# Amiblitz3 at 58a11658: its token list, read like blitz2's) with the same
# libraries.  The library tokens are numbered by blitztokens for both
# (AmiBlitz3's process_libraries numbers as blitz2's fixlibs).  Skipped
# (77) without the fetched files.
set(AB3_V314 ${PROJECT_SOURCE_DIR}/archive/amiblitz3/github-58a11658)
set(ab3_biglibs baseLibs utilLibs OsLibs csLibs)
list(TRANSFORM ab3_biglibs PREPEND ${AB3_V314}/BlitzLibs/)
# ab3_tokens(NAME LAST): the test with blitz2-ab3's list ending with $LAST
function(ab3_tokens name last)
	add_test(NAME ${name} COMMAND ${CMAKE_COMMAND} -DBLITZTOKENS=${BLITZTOKENS}
		-DBLITZ2=$<TARGET_FILE:blitz2-ab3> -DLAST=${last} -DAB3=${AB3_V314}/Amiblitz3
		"-DLIBS=${ab3_biglibs}" -DOUT=${CMAKE_CURRENT_BINARY_DIR}/${name}
		-P ${CMAKE_CURRENT_SOURCE_DIR}/tokens.cmake)
	set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77)
endfunction()
ab3_tokens(ab3_tokens 80DB)

# expect file - or with cli/NAME.ref.expect, a difference of AmiBlitz3

	"ab3 command line tests: time limit of the emulator session (seconds)")
# the sources of git, and those made in the build: the tokenised includes
# (tedtok, BB2's token list) and the AmiBlitz3 .xtra of one of them
file(GLOB cli_sources CONFIGURE_DEPENDS ${cli_dir}/*.ab3 ${cli_dir}/*.bb)
set(ted_tokens ${PROJECT_SOURCE_DIR}/tests/data/ted-tokens.txt)
set(cli_made)
foreach(pair incerr_tok_inc|incerr_tok_inc incother|incother_bb2tok incother|incother_ab3tok)
	string(REPLACE "|" ";" pair "${pair}")
	list(GET pair 0 in)
	list(GET pair 1 out)
	add_custom_command(OUTPUT ${cli_gen}/${out}.bb2
		COMMAND ${CMAKE_COMMAND} -E make_directory ${cli_gen}
		COMMAND ${TEDTOK} ${ted_tokens} ${cli_dir}/${in}.bas ${cli_gen}/${out}.bb2
		DEPENDS ${cli_dir}/${in}.bas ${ted_tokens} ${TEDTOK} host_tools
		COMMENT "Tokenising tests/ab3/cli/${in}.bas as ${out}.bb2"
		VERBATIM)
	list(APPEND cli_made ${cli_gen}/${out}.bb2)
endforeach()
add_custom_command(OUTPUT ${cli_gen}/incother_ab3tok.bb2.xtra
	COMMAND ${CMAKE_COMMAND} -E make_directory ${cli_gen}
	COMMAND ${MKTESTDATA} ${cli_dir}/ab3xtra.spec ${cli_gen}/incother_ab3tok.bb2.xtra
	DEPENDS ${cli_dir}/ab3xtra.spec ${MKTESTDATA} host_tools
	COMMENT "AmiBlitz3 .xtra of incother_ab3tok.bb2"
	VERBATIM)
list(APPEND cli_made ${cli_gen}/incother_ab3tok.bb2.xtra)

# ab3_cli_case(NAME SOURCE [TREE Blitz3|Blitz3-nolib|Blitz3-userlib]
#              [ARGS ARG...] [REF] [REF_EXPECT FILE] [SAME CASE]): a
#              case of the suite; SAME: its executable is the one of the
#              case CASE, in both configurations
set(cli_exe_cases "")
set(cli_ref_cases "")
set(cli_exe_same "")
set(cli_ref_same "")
macro(ab3_cli_case name source)
	cmake_parse_arguments(c "REF" "TREE;REF_EXPECT;SAME" "ARGS" ${ARGN})
	if(NOT c_TREE)
		set(c_TREE Blitz3)
	endif()
	list(JOIN c_ARGS " " c_args)
	set(c_line "${name}|${c_TREE}|${source}|${c_args}|${cli_dir}/${name}.expect")
	string(APPEND cli_exe_cases "${c_line}\n")
	if(c_SAME)
		list(APPEND cli_exe_same "${name}=${c_SAME}")
	endif()
	if(c_REF)
		if(c_REF_EXPECT)
			string(APPEND c_line "|${cli_dir}/${c_REF_EXPECT}")
		endif()
		string(APPEND cli_ref_cases "${c_line}\n")
		if(c_SAME)
			list(APPEND cli_ref_same "${name}=${c_SAME}")
		endif()
	endif()
endmacro()

# ab3_cli(NAME CONFIG CASES [SAME LIST]): the ctest
function(ab3_cli name config cases)
	cmake_parse_arguments(a "" "" "SAME" ${ARGN})
	list(JOIN a_SAME "," same)
	set(work ${CMAKE_CURRENT_BINARY_DIR}/${name})
	file(WRITE ${work}.cases "${cases}")
	set(sources ${cli_sources} ${cli_made})
	string(REPLACE ";" "$<SEMICOLON>" sources "${sources}")
	add_test(NAME ${name} COMMAND ${CMAKE_COMMAND} -DAMIGA_RUN=${AMIGA_RUN}
		-DOUTMATCH=${OUTMATCH} -DREQTRAP=$<TARGET_FILE:reqtrap>
		-DCONFIG=${config} -DCASES=${work}.cases
		-DBLITZ2_DIR=${BLITZ2_DIR} -DAB3=${AB3_V314} -DSOURCES=${sources}
		-DUSERLIB=${CMAKE_CURRENT_BINARY_DIR}/amiga/ab3_userlib.obj
		-DWORK=${work} -DTIMEOUT=${AB3_CLI_TIMEOUT}
		-DSAME=${same}
		-P ${CMAKE_CURRENT_SOURCE_DIR}/cli.cmake)
	math(EXPR limit "${AB3_CLI_TIMEOUT} + 120 + ${EMULATION_QUEUE_WAIT}")
	set_tests_properties(${name} PROPERTIES LABELS emulator SKIP_RETURN_CODE 77
		TIMEOUT ${limit})
endfunction()
