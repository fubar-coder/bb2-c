# The tests of blitz2-ab3 (docs/ab3/variant.md).
#
# Equivalence with blitz2 (BB2 mode): every source of the corpus
# (tests/corpus/tests*/), of the bug-orig tests of the old form
# (tests/bug-orig/*/), the driver of every test of the BASIC test model
# that the prev build compares (the bug-orig, regression, blitz2 and
# expression matrix tests) and the data sources of the expression matrix's
# tests of refused cells is compiled through tedstub once by blitz2 (the
# reference) and once by blitz2-ab3, with the C libraries of this build,
# the bug-orig user library and the test library in blitz2:userlibs; the
# programs are not run (corpusrun run=off).  Every source must give
# the same compile result (tedstub's RESULT line: size, or error line
# and message) and the same executable - byte-identical without runtime
# error checking, with it the trap #1/#4 parameter blocks zeroed
# (the address of the editor's line item depends on the memory layout).
# A bug-orig source that needs more than that
# (RWE userlibs, its own jobs or residents) does not compile in either:
# it is compared by its error; one that cannot be compared is left out
# (below).  Nothing of the CD (the executables never contain
# blitz2:defaultdbug, the sources come from the repository and the build):
# label emulator, skipped (77) only without Amiberry.
get_property(corpus_sources GLOBAL PROPERTY CORPUS_SOURCES)
get_property(bug_orig_sources GLOBAL PROPERTY BUG_ORIG_SOURCES)
get_property(bug_orig_userlib GLOBAL PROPERTY BUG_ORIG_USERLIB)
# the drivers of the tests of the BASIC test model compared with the prev build
# (cmake/basic-test-model.cmake: PATH|STEM/ID) and the data sources of the
# expression matrix's tests of refused cells (PATH|ID.test/NAME.bas), with the
# test library in blitz2:userlibs
get_property(model_drivers GLOBAL PROPERTY BASIC_TEST_DRIVERS)
get_property(exprmatrix_data GLOBAL PROPERTY EXPRMATRIX_DATA_SOURCES)
# not compared - NAME|reason:
#   compiler-maxs-libjsr      the unresolved JSRs inside linked library code
#                             (#335) put addresses of the compiler's memory
#                             into the executable: they depend on its layout
foreach(n compiler-maxs-libjsr)
	list(FILTER bug_orig_sources EXCLUDE REGEX "/${n}\\.bb2$")
	list(FILTER model_drivers EXCLUDE REGEX "/${n}\\.bb2\\|")
endforeach()
set(AB3_EQUIVALENCE_TIMEOUT 600 CACHE STRING
	"ab3 equivalence: time limit of one emulator session (seconds)")

# equivalence_pass(NAME on|off): pass file + ctest
function(equivalence_pass name runtime_errors)
	set(work ${CMAKE_CURRENT_BINARY_DIR}/${name})
	set(text "# ${name}: generated by tests/ab3/CMakeLists.txt\n")
	string(APPEND text "work=${work}\nreport=${work}.txt\nrun=off\n"
		"runtime_errors=${runtime_errors}\ntimeout=${AB3_EQUIVALENCE_TIMEOUT}\n"
		"parallel=${CORPUS_PARALLEL}\nted=$<TARGET_FILE:tedstub>\n"
		"config=blitz2|ref|$<TARGET_FILE:blitz2>|${BLITZ2_DIR}/deflibs|${BLITZ2_DIR}/acidlibs\n"
		"config=ab3|exe|$<TARGET_FILE:blitz2-ab3>|${BLITZ2_DIR}/deflibs|${BLITZ2_DIR}/acidlibs\n"
		"file=blitz2/userlibs/bugorig/bugorig.obj|${bug_orig_userlib}\n"
		"file=blitz2/userlibs/testlib/testlib.obj|${TESTLIB_OBJ}\n")
	foreach(s IN LISTS corpus_sources bug_orig_sources)
		string(APPEND text "source=${s}\n")
	endforeach()
	foreach(s IN LISTS model_drivers exprmatrix_data)
		string(APPEND text "source=${s}\n")
	endforeach()
	file(GENERATE OUTPUT ${work}.pass CONTENT "${text}")
	add_test(NAME ${name} COMMAND ${CORPUSRUN} pass ${work}.pass)
	# two sessions, maybe one after the other, each maybe waiting for a slot
	math(EXPR limit "${AB3_EQUIVALENCE_TIMEOUT} * 2 + 120 + 2 * ${EMULATION_QUEUE_WAIT}")
	set_tests_properties(${name} PROPERTIES LABELS emulator SKIP_RETURN_CODE 77
		TIMEOUT ${limit})
endfunction()

# ab3_cli_case(NAME SOURCE [TREE Blitz3|Blitz3-nolib|Blitz3-nobase|Blitz3-userlib|none]
#              [ARGS ARG...] [REF] [REF_EXPECT FILE] [REF_NORUN] [DIR SUBDIR]
#              [NORUN] [SAME CASE] [HUNKS LAYOUT] [DATESTAMP ISO-DATE]
#              [RUN_DATESTAMP ISO-DATE]): a case of the suite (a case that
#              only checks the program's output without other arguments
#              than -r is a test of tests/ab3/model/ instead, change
#              ab3-test-model); DATESTAMP: the
#              compiler runs under inject datestamp=DATE (cli.cmake:
#              dos.library's DateStamp gives that date, so a compile-time
#              date is fixed), RUN_DATESTAMP: the same for the program (its
#              SystemDate), in both configurations; SAME:
#              its executable is the one of the case CASE, HUNKS: its hunk
#              layout (hunkcheck layout, e.g. code,bss,data+chip), in both
#              configurations; REF_NORUN: AmiBlitz3's executable is kept
#              but not run (a broken executable that damages the session,
#              docs/ab3/variant.md), its [run] is compared with
#              cli/norun.ref.expect; NORUN: neither executable is run (a
#              program that needs hardware the session does not have:
#              compile only), [run] is "not run" in both; DIR: the expect
#              files (NAME.expect, REF_EXPECT) are in cli/SUBDIR
set(cli_exe_cases "")
set(cli_ref_cases "")
set(cli_exe_same "")
set(cli_ref_same "")
set(cli_exe_hunks "")
set(cli_ref_hunks "")
macro(ab3_cli_case name source)
	cmake_parse_arguments(c "REF;REF_NORUN;NORUN" "TREE;REF_EXPECT;SAME;HUNKS;DATESTAMP;RUN_DATESTAMP;DIR" "ARGS" ${ARGN})
	set(c_dir ${cli_dir})
	if(c_DIR)
		set(c_dir ${cli_dir}/${c_DIR})
	endif()
	set(c_datestamp)
	if(c_DATESTAMP)
		if(NOT c_DATESTAMP MATCHES "^[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9]$")
			message(FATAL_ERROR "ab3_cli_case ${name}: DATESTAMP ${c_DATESTAMP} is not YYYY-MM-DD")
		endif()
		set(c_datestamp "|datestamp=${c_DATESTAMP}")
	endif()
	if(c_RUN_DATESTAMP)
		if(NOT c_RUN_DATESTAMP MATCHES "^[0-9][0-9][0-9][0-9]-[0-9][0-9]-[0-9][0-9]$")
			message(FATAL_ERROR "ab3_cli_case ${name}: RUN_DATESTAMP ${c_RUN_DATESTAMP} is not YYYY-MM-DD")
		endif()
		string(APPEND c_datestamp "|rundatestamp=${c_RUN_DATESTAMP}")
	endif()
	if(NOT c_TREE)
		set(c_TREE Blitz3)
	endif()
	list(JOIN c_ARGS " " c_args)
	set(c_line "${name}|${c_TREE}|${source}|${c_args}|${c_dir}/${name}.expect")
	if(c_NORUN)
		string(APPEND cli_exe_cases "${c_line}|${c_dir}/${name}.expect|norun${c_datestamp}\n")
		set(c_REF_NORUN ON)
	else()
		string(APPEND cli_exe_cases "${c_line}${c_datestamp}\n")
	endif()
	if(c_SAME)
		list(APPEND cli_exe_same "${name}=${c_SAME}")
	endif()
	if(c_HUNKS)
		string(APPEND cli_exe_hunks "|${name}=${c_HUNKS}")
	endif()
	if(c_REF)
		if(c_REF_EXPECT)
			string(APPEND c_line "|${c_dir}/${c_REF_EXPECT}")
		endif()
		if(c_REF_NORUN)
			if(c_REF_EXPECT)
				message(FATAL_ERROR "ab3_cli_case ${name}: REF_NORUN and REF_EXPECT")
			endif()
			string(APPEND c_line "|${cli_dir}/norun.ref.expect|norun")
		endif()
		string(APPEND cli_ref_cases "${c_line}${c_datestamp}\n")
		if(c_SAME)
			list(APPEND cli_ref_same "${name}=${c_SAME}")
		endif()
		if(c_HUNKS)
			string(APPEND cli_ref_hunks "|${name}=${c_HUNKS}")
		endif()
	endif()
endmacro()

# ab3_cli(NAME CONFIG CASES [SAME LIST] [HUNKS |NAME=LAYOUT...]): the ctest
function(ab3_cli name config cases)
	cmake_parse_arguments(a "WANT_FAIL" "HUNKS" "SAME" ${ARGN})
	# WANT_FAIL: the test is about a case that must fail (its output is matched): with-workdir.cmake
	# removes the directory of that failure
	if(a_WANT_FAIL)
		set(a_WANT_FAIL -DWANT_FAIL=ON)
	else()
		set(a_WANT_FAIL "")
	endif()
	string(REGEX REPLACE "^\\|" "" hunks "${a_HUNKS}")
	list(JOIN a_SAME "," same)
	set(work ${CMAKE_CURRENT_BINARY_DIR}/${name})
	file(WRITE ${work}.cases "${cases}")
	set(sources ${cli_sources} ${cli_made} ${ab3_dev_sources})
	string(REPLACE ";" "$<SEMICOLON>" sources "${sources}")
	add_test(NAME ${name} COMMAND ${CMAKE_COMMAND} -DAMIGA_RUN=${AMIGA_RUN}
		-DRESULTS=${RESULTS} -DREQTRAP=$<TARGET_FILE:reqtrap> -DINJECT=$<TARGET_FILE:inject>
		-DCONFIG=${config} -DCASES=${work}.cases
		-DBLITZ2_DIR=${BLITZ2_DIR} -DAB3=${AB3_V314} -DSOURCES=${sources}
		-DUSERLIB=${CMAKE_CURRENT_BINARY_DIR}/amiga/ab3_userlib.obj
		-DWORK=${work} -DTIMEOUT=${AB3_CLI_TIMEOUT}
		-DSAME=${same} -DHUNKCHECK=${HUNKCHECK} "-DHUNKS=${hunks}"
		-DTEST_SCRIPT=${CMAKE_CURRENT_SOURCE_DIR}/cli.cmake ${a_WANT_FAIL} -P ${WITH_WORKDIR})
	math(EXPR limit "${AB3_CLI_TIMEOUT} + 120 + ${EMULATION_QUEUE_WAIT}")
	set_tests_properties(${name} PROPERTIES LABELS emulator SKIP_RETURN_CODE 77
		TIMEOUT ${limit})
	# with a prev build: the block's cases compiled by the previous commit's blitz2-ab3 as well,
	# the executables and the compiler's output must be the same (cli-prev.cmake, ab3_cli_prev_N)
	if(PREV_REQUESTED AND config STREQUAL "exe" AND name MATCHES "^ab3_cli_[0-9]+$")
		string(REGEX REPLACE "^ab3_cli_" "ab3_cli_prev_" prev_name ${name})
		set(prev_intended "${PREV_INTENDED}")
		string(REPLACE ";" "$<SEMICOLON>" prev_intended "${prev_intended}")
		prev_register(ab3_cli_prev)
		add_test(NAME ${prev_name} COMMAND ${CMAKE_COMMAND} -DAMIGA_RUN=${AMIGA_RUN}
			-DRESULTS=${RESULTS} -DREQTRAP=$<TARGET_FILE:reqtrap> -DINJECT=$<TARGET_FILE:inject>
			-DCASES=${work}.cases -DBLITZ2_DIR=${BLITZ2_DIR} -DPREV_BLITZ2_DIR=${PREV_BLITZ2_DIR}
			-DAB3=${AB3_V314} -DSOURCES=${sources}
			-DUSERLIB=${CMAKE_CURRENT_BINARY_DIR}/amiga/ab3_userlib.obj
			-DWORK=${work}-prev -DTIMEOUT=${AB3_CLI_TIMEOUT} "-DINTENDED=${prev_intended}"
			-DTEST_SCRIPT=${CMAKE_CURRENT_SOURCE_DIR}/cli-prev.cmake -P ${WITH_WORKDIR})
		# two sessions at the same time
		math(EXPR limit "${AB3_CLI_TIMEOUT} + 120 + 2 * ${EMULATION_QUEUE_WAIT}")
		set_tests_properties(${prev_name} PROPERTIES LABELS emulator SKIP_RETURN_CODE 77
			TIMEOUT ${limit})
	endif()
endfunction()
# ab3_cli_blocks(NAME CONFIG CASES OUT [SAME LIST] [HUNKS |NAME=LAYOUT...]): the
# cases of CASES split into blocks of at most AB3_CLI_BLOCK cases, spread
# evenly, in their order: the ctests NAME_1 ... NAME_N, one session each.
# A SAME pair compares the executables of two cases of one session, so a
# block does not end between its two cases (it grows beyond the size
# instead); every block gets the SAME pairs and the HUNKS layouts of its
# own cases.  The tests of the blocks are returned in OUT.
set(AB3_CLI_BLOCK_DEFAULT 70)
if(BLITZ2_COVERAGE)
	# the coverage compilers take about 5 s per case
	set(AB3_CLI_BLOCK_DEFAULT 20)
endif()
set(AB3_CLI_BLOCK ${AB3_CLI_BLOCK_DEFAULT} CACHE STRING
	"ab3 command line tests: the most cases of one emulator session (ab3_cli_N, ab3_cli_ref_N; default 70, 20 with BLITZ2_COVERAGE)")
if(NOT AB3_CLI_BLOCK MATCHES "^[1-9][0-9]*$")
	message(FATAL_ERROR "AB3_CLI_BLOCK=${AB3_CLI_BLOCK} is not a number of cases")
endif()
function(ab3_cli_blocks name config cases out)
	cmake_parse_arguments(a "" "HUNKS" "SAME" ${ARGN})
	string(REGEX REPLACE "\n$" "" lines "${cases}")
	string(REPLACE "\n" ";" lines "${lines}")
	set(names)
	foreach(l IN LISTS lines)
		string(REGEX REPLACE "\\|.*" "" n "${l}")
		list(APPEND names ${n})
	endforeach()
	list(LENGTH names count)
	# every case of a SAME pair is in the list
	foreach(p IN LISTS a_SAME)
		string(REPLACE "=" ";" ab "${p}")
		foreach(n IN LISTS ab)
			list(FIND names ${n} i)
			if(i LESS 0)
				message(FATAL_ERROR "${name}: SAME ${p}: no case ${n}")
			endif()
		endforeach()
	endforeach()
	string(REGEX REPLACE "^\\|" "" hunks "${a_HUNKS}")
	string(REPLACE "|" ";" hunks "${hunks}")
	math(EXPR blocks "(${count} + ${AB3_CLI_BLOCK} - 1) / ${AB3_CLI_BLOCK}")
	math(EXPR size "(${count} + ${blocks} - 1) / ${blocks}")
	set(tests)
	set(k 0)
	set(first 0)
	while(first LESS count)
		# the end of the block: SIZE cases, more while a SAME pair has one
		# case inside and one behind
		math(EXPR last "${first} + ${size} - 1")
		set(grown TRUE)
		while(grown AND last LESS count)
			set(grown FALSE)
			foreach(p IN LISTS a_SAME)
				string(REPLACE "=" ";" ab "${p}")
				list(GET ab 0 pa)
				list(GET ab 1 pb)
				list(FIND names ${pa} ia)
				list(FIND names ${pb} ib)
				if((ia LESS_EQUAL last AND ib GREATER last) OR (ib LESS_EQUAL last AND ia GREATER last))
					math(EXPR last "${last} + 1")
					set(grown TRUE)
				endif()
			endforeach()
		endwhile()
		if(last GREATER_EQUAL count)
			math(EXPR last "${count} - 1")
		endif()
		set(text "")
		set(bnames)
		foreach(i RANGE ${first} ${last})
			list(GET lines ${i} l)
			string(APPEND text "${l}\n")
			list(GET names ${i} n)
			list(APPEND bnames ${n})
		endforeach()
		set(bsame)
		foreach(p IN LISTS a_SAME)
			string(REPLACE "=" ";" ab "${p}")
			list(GET ab 0 pa)
			if(pa IN_LIST bnames)
				list(APPEND bsame ${p})
			endif()
		endforeach()
		set(bhunks "")
		foreach(h IN LISTS hunks)
			string(REGEX REPLACE "=.*" "" n "${h}")
			if(n IN_LIST bnames)
				string(APPEND bhunks "|${h}")
			endif()
		endforeach()
		math(EXPR k "${k} + 1")
		ab3_cli(${name}_${k} ${config} "${text}" SAME ${bsame} HUNKS "${bhunks}")
		list(APPEND tests ${name}_${k})
		math(EXPR first "${last} + 1")
	endwhile()
	set(${out} ${tests} PARENT_SCOPE)
endfunction()

# AB3 mode's tokens against AmiBlitz3's (spec ab3-compat/tokenizer,
# tokenise.cmake): AmiBlitz3 v3.14.0 loads every source in CLI mode and
# writes its lines (-save NAME.bb2), bbsrc ab3-cmp (shared/tokenizer, as
# ab3cli) loads it with AB3 mode's token list, the lines must be equal.  Sources:
# AmiBlitz3's test programs, those of cli/ and tokens/ (edge cases).
# Label emulator; skipped (77) without Amiberry or the fetched files.
file(GLOB tokenise_sources CONFIGURE_DEPENDS ${cli_dir}/*.ab3
	${CMAKE_CURRENT_SOURCE_DIR}/tokens/*.ab3)
# not compared: esc_repeat_quote ("\\R03" before the closing quote) -
# AmiBlitz3 writes its lines, but a later -save of the session ends in a
# program error (8000000A), as if the line damaged its memory
list(FILTER tokenise_sources EXCLUDE REGEX "/esc_repeat_quote\\.ab3$")
string(REPLACE ";" "$<SEMICOLON>" tokenise_sources "${tokenise_sources}")
# ab3_tokenise(NAME [BREAK]): the ctest; BREAK: the token of NPrint
# changed in our list (the check itself)
function(ab3_tokenise name)
	set(work ${CMAKE_CURRENT_BINARY_DIR}/${name})
	set(brk)
	if(ARGN)
		set(brk -DBREAK=NPrint)
	endif()
	add_test(NAME ${name} COMMAND ${CMAKE_COMMAND} -DAMIGA_RUN=${AMIGA_RUN}
		-DBBSRC=${BBSRC} -DBLITZ2=$<TARGET_FILE:blitz2-ab3>
		-DAB3=${AB3_V314} -DSOURCES=${tokenise_sources} -DWORK=${work} ${brk}
		-DTIMEOUT=${AB3_CLI_TIMEOUT} -P ${CMAKE_CURRENT_SOURCE_DIR}/tokenise.cmake)
	math(EXPR limit "${AB3_CLI_TIMEOUT} + 120 + ${EMULATION_QUEUE_WAIT}")
	set_tests_properties(${name} PROPERTIES LABELS emulator SKIP_RETURN_CODE 77
		TIMEOUT ${limit})
endfunction()
