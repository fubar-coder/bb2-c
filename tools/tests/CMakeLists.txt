# tests of the host tools, with broken inputs built in memory (testutil.h)
foreach(t hunk blitzlibcheck hunkcheck)
	add_executable(test_${t} test_${t}.c)
	target_compile_options(test_${t} PRIVATE -Wall -Wextra -Wno-unused-function)
	add_test(NAME ${t} COMMAND test_${t})
endforeach()
target_link_libraries(test_hunk PRIVATE hunk)
target_link_libraries(test_blitzlibcheck PRIVATE blitzlib)
target_link_libraries(test_hunkcheck PRIVATE hunkchecklib)
