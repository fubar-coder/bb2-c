# testinfo: records of test runs (README.md)
# rules.txt: what a change of a file selects (select; cmake/coverage-select.cmake)
set(TESTINFO_SUBCOMMANDS stats map select)
add_executable(testinfo main.cpp cmd_stats.cpp cmd_map.cpp cmd_select.cpp)
target_link_libraries(testinfo PRIVATE lib_cli lib_teststats lib_covmap lib_covsel)
set_target_properties(testinfo PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
host_help_test(testinfo CLI ${TESTINFO_SUBCOMMANDS})
host_readme_test(testinfo README.md CLI ${TESTINFO_SUBCOMMANDS})
